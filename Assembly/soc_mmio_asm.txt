# G7 MIPS32 - MMIO and data RAM demo
# Assembly source for soc_mmio.mem. Each line in the .mem file is one
# 32-bit instruction. The first instruction is at byte address 0x00.
#
# PC    Instruction                    Machine code   What it does

lui  $t1, 0x8000                     # 0x00  3C098000  $t1 = 0x80000000 (ParIO)
addi $t0, $zero, 5                   # 0x04  20080005  $t0 = 5
sw   $t0, 0($t1)                     # 0x08  AD280000  Write 5 to ParIO
lw   $t2, 0($t1)                     # 0x0C  8D2A0000  Read the buttons and switches
addi $t3, $zero, 42                  # 0x10  200B002A  $t3 = 42
sw   $t3, 4($zero)                   # 0x14  AC0B0004  Write 42 to RAM at byte address 4
lw   $t4, 4($zero)                   # 0x18  8C0C0004  Read 42 back into $t4
done: j done                        # 0x1C  08000007  Stay at this address

# Why the same LW and SW instructions reach different devices:
#   The CPU calculates base + offset. The data bus sends address
#   0x80000000 to ParIO and aligned byte address 0x00000004 to data RAM.
#   Address 4 is RAM word index 1 because each word occupies four bytes.
#
# Expected results:
#   - ParIO takes the low three bits of 5 (101) and drives LD3:LD1.
#     LD0 follows the reset input. With reset released, o_led is 1010.
#   - With testbench inputs BTN3:BTN1 = 101 and SW3:SW0 = 1010,
#     the ParIO read packs them as 0_101_1010 = 0x5A. The load puts
#     0x0000005A in $t2. On the board, this value depends on the inputs.
#   - RAM word 1 and $t4 contain 42 (0x0000002A).
#   - The final jump loops without issuing any more memory requests.
#
# The behavioral tb_soc_mmio checks LEDs, registers, and bus requests.
# The port-only tb_soc_timing checks the LEDs with routed FPGA delays.
# For that timing test, select soc_mmio.mem as the SoC ROM image BEFORE
# synthesis so the program is built into the netlist.
#
# To regenerate the ROM image from this text using the project assembler:
#   python3 Assembly/asm.py Assembly/soc_mmio_asm.txt -o Assembly/soc_mmio.mem
